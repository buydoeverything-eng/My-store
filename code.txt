        init() {
          const savedCart = localStorage.getItem("buydo-cart");

          if (savedCart) {
            try {
              this.cart = JSON.parse(savedCart);
            } catch (error) {
              this.cart = [];
            }
          }
        },


        saveCart() {
          localStorage.setItem(
            "buydo-cart",
            JSON.stringify(this.cart)
          );
        },


        addToCart(product) {

          const existingItem = this.cart.find(
            item => item.id === product.id
          );

          if (existingItem) {
            existingItem.quantity++;
          } else {
            this.cart.push({
              ...product,
              quantity: 1
            });
          }

          this.saveCart();

          this.cartOpen = true;
        },


        removeFromCart(id) {

          this.cart = this.cart.filter(
            item => item.id !== id
          );

          this.saveCart();
        },


        increaseQuantity(id) {

          const item = this.cart.find(
            item => item.id === id
          );

          if (item) {
            item.quantity++;
          }

          this.saveCart();
        },


        decreaseQuantity(id) {

          const item = this.cart.find(
            item => item.id === id
          );

          if (!item) return;

          if (item.quantity > 1) {
            item.quantity--;
          } else {
            this.removeFromCart(id);
            return;
          }

          this.saveCart();
        },


        get cartCount() {

          return this.cart.reduce(
            (total, item) => total + item.quantity,
            0
          );

        },


        get cartTotal() {

          return this.cart.reduce(
            (total, item) =>
              total + (item.price * item.quantity),
            0
          );

        },

            style: "currency",
            currency: "INR",
            maximumFractionDigits: 0
          }).format(price);

        },
